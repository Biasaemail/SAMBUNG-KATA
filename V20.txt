local FOLDER_NAME = "WORD"

local function ensureDirectory()

-- Fungsi Save/Load JSON
local function saveSettings()
    local data = {
        TargetSuffixRaw = App.Config.TargetSuffixRaw,
        TypingDelayMS = App.Config.TypingDelayMS,
        AutoPlay = App.Config.AutoPlay
    }
    local success, encoded = pcall(function() return HttpService:JSONEncode(data) end)
    if success then
        writefile(CONFIG_FILE, encoded)
    end
end

local function loadSettings()
    if isfile(CONFIG_FILE) then
        local success, content = pcall(function() return readfile(CONFIG_FILE) end)
        if success then
            local ok, decoded = pcall(function() return HttpService:JSONDecode(content) end)
            if ok then
                App.Config.TargetSuffixRaw = decoded.TargetSuffixRaw or ""
                App.Config.TypingDelayMS = decoded.TypingDelayMS or 250
                App.Config.AutoPlay = decoded.AutoPlay or true
                
                -- Re-process suffixes
                App.Config.TargetSuffixes = {}
                for sfx in string.gmatch(App.Config.TargetSuffixRaw, "([^,]+)") do
                    local cleanSfx = lower(sfx:match("^%s*(.-)%s*$") or "")
                    if cleanSfx ~= "" then table.insert(App.Config.TargetSuffixes, cleanSfx) end
                end
            end
        end
    end
end

-- Load settings di awal
loadSettings()

local function saveNewWords()
    if #App.DB.NewWordsQueue > 0 then
        local data = table.concat(App.DB.NewWordsQueue, "\n") .. "\n"
        if appendfile then 
            pcall(function() appendfile(BANKWORD_FILE, data) end)
        end
        table.clear(App.DB.NewWordsQueue)
    end
end
task.spawn(function() while true do task.wait(5); saveNewWords() end end)

local function addWordToDB(word, saveToBank)
    local lw = lower(word:match("^%s*(%a+)%s*$") or "")
    if not lw or #lw < 2 or App.State.PermanentBlacklist[lw] or App.DB.KnownWords[lw] then return false end

    App.DB.KnownWords[lw] = true
    App.DB.TotalWords = App.DB.TotalWords + 1
    App.DB.Dictionary[App.DB.TotalWords] = lw

    local len = #lw
    for i = 1, math.min(4, len) do
        local prefix = sub(lw, 1, i)
        if not App.DB.PrefixMap[prefix] then App.DB.PrefixMap[prefix] = {} end
        table.insert(App.DB.PrefixMap[prefix], App.DB.TotalWords)
    end

    local firstLetter = sub(lw, 1, 1)
    App.DB.WordsStartingWith[firstLetter] = (App.DB.WordsStartingWith[firstLetter] or 0) + 1

    if saveToBank then table.insert(App.DB.NewWordsQueue, lw) end
    return true
end
